# pinned so versions are decided up front rather than whatever pip resolves later.

# day 2: chunking + dense retrieval.
# sentence-transformers pulls torch (~200 MB), which is the bulk of a fresh install.
# transformers is imported directly in ingest/chunk.py for the wordpiece counter.
# it arrives via sentence-transformers anyway, but a direct import is a direct dependency.
transformers==4.44.2
sentence-transformers==3.0.1
faiss-cpu==1.8.0

# numpy and torch are imported directly by retrieval/dense.py and retrieval/rerank.py, so
# the rule three lines up applies to them too. Both were missing here until 2026-08-08.
#
# The cap is not housekeeping. faiss-cpu 1.8.0 ships compiled against numpy 1.x and asks
# for numpy with no upper bound. A clean install therefore resolves numpy 2.x and the very
# first `import faiss` dies with "_ARRAY_API not found". Reproduced on 2026-08-08 against
# numpy 2.2.6, which is what pip resolved for this file on that day.
numpy<2

# torch is deliberately not pinned. The version this project's latency figures were
# measured on was never written down. Choosing one now would hand those figures a
# provenance they do not have, which is worse than an unpinned line.
# On a CPU only machine install it from the cpu index first. The default PyPI wheel
# carries CUDA and is far larger.
#   pip install torch --index-url https://download.pytorch.org/whl/cpu
torch
